cmake_minimum_required(VERSION 3.16)
project(codex_hud_preview LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
get_filename_component(FIRMWARE "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
set(LV_BUILD_CONF_PATH "${FIRMWARE}/include/lv_conf.h" CACHE PATH "" FORCE)
set(CONFIG_LV_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_BUILD_DEMOS OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_USE_THORVG OFF CACHE BOOL "" FORCE)
add_subdirectory("${FIRMWARE}/.pio/libdeps/esp32s3_nano/lvgl" lvgl)
add_executable(hud-preview main.cpp "${FIRMWARE}/src/lvgl_ui.cpp" "${FIRMWARE}/src/font_cn_16.c" "${FIRMWARE}/src/font_cn_20.c" "${FIRMWARE}/src/font_cn_24.c")
target_include_directories(hud-preview PRIVATE "${CMAKE_CURRENT_LIST_DIR}" "${FIRMWARE}/src" "${FIRMWARE}/include")
target_compile_definitions(hud-preview PRIVATE LV_LVGL_H_INCLUDE_SIMPLE)
target_link_libraries(hud-preview PRIVATE lvgl)
